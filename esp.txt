#include <Arduino.h>
#include "HX711.h"
#include <Wire.h>
#include "Adafruit_SHT31.h"
#include <MPU9250_asukiaaa.h>
#include <WiFi.h>
#include <HTTPClient.h>

// ================= HX711 =================
#define LOADCELL_DOUT_PIN 4
#define LOADCELL_SCK_PIN 26

HX711 scale;

float calibration_factor = 2280.0;

float filteredWeight = 0;
float alpha = 0.3;
float lastWeight = 0;

// ================= RELAY =================
#define RELAY_PIN 2

bool extractionON = false;

// ================= BLUETOOTH =================
#define RXD2 16
#define TXD2 17

HardwareSerial hc06(2);

// ================= SHT31 =================
Adafruit_SHT31 sht31 = Adafruit_SHT31();

#define SHT31_ADDR 0x44

// ================= MPU9250 =================
MPU9250_asukiaaa IMU;

// ================= WIFI =================
const char* ssid = "Airbox_TEST";
const char* password = "Wifi@1234567";

// ===== FLASK SERVER =====
const char* sensorURL =
"http://192.168.1.200:5000/sensors";
const char* controlURL =
"http://192.168.1.200:5000/control/status";

// =====================================================
// ================= WIFI CONNECT ======================
// =====================================================
void connectWiFi() {

  WiFi.begin(ssid, password);

  Serial.print("Connexion WiFi");

  int tries = 0;

  while (WiFi.status() != WL_CONNECTED && tries < 20) {

    delay(500);
    Serial.print(".");
    tries++;
  }

  if (WiFi.status() == WL_CONNECTED) {

    Serial.println("\n✅ WiFi connecté !");
    Serial.println(WiFi.localIP());

  } else {

    Serial.println("\n❌ WiFi échoué !");
  }
}

// =====================================================
// ================= GET EXTRACTION STATUS =============
// =====================================================
void getExtractionStatus() {

  if (WiFi.status() == WL_CONNECTED) {

    HTTPClient http;

    http.begin(controlURL);

    int code = http.GET();

    if (code > 0) {

      String payload = http.getString();

      Serial.println(payload);

      // ===== STATUS TRUE =====
      if (payload.indexOf("true") > 0) {

        extractionON = true;

      } else {

        extractionON = false;
      }

      // ===== RELAY =====
      if (extractionON) {

        digitalWrite(RELAY_PIN, HIGH);

        Serial.println("🟢 EXTRACTION ACTIVE");

      } else {

        digitalWrite(RELAY_PIN, LOW);

        Serial.println("🔴 EXTRACTION STOP");
      }
    }

    http.end();
  }
}

// =====================================================
// ================= SEND SENSOR DATA ==================
// =====================================================
void sendSensorData(
  float weight,
  float temp,
  float hum,
  float ax,
  float ay,
  float az
) {

  if (WiFi.status() == WL_CONNECTED) {

    HTTPClient http;

    http.begin(sensorURL);

    http.addHeader(
      "Content-Type",
      "application/json"
    );

    // ===== JSON =====
    String jsonData = "{";

    jsonData += "\"weight\":" + String(weight) + ",";
    jsonData += "\"temperature\":" + String(temp) + ",";
    jsonData += "\"humidity\":" + String(hum) + ",";
    jsonData += "\"ax\":" + String(ax) + ",";
    jsonData += "\"ay\":" + String(ay) + ",";
    jsonData += "\"az\":" + String(az);

    jsonData += "}";

    // ===== SEND =====
    int code = http.POST(jsonData);

    if (code > 0) {

      Serial.print("📡 Données envoyées : ");
      Serial.println(code);

    } else {

      Serial.print("❌ Erreur HTTP : ");
      Serial.println(code);
    }

    http.end();
  }
}

// =====================================================
// ================= SETUP =============================
// =====================================================
void setup() {

  Serial.begin(115200);

  // ===== RELAY =====
  pinMode(RELAY_PIN, OUTPUT);

  digitalWrite(RELAY_PIN, LOW);

  // ===== BLUETOOTH =====
  hc06.begin(9600, SERIAL_8N1, RXD2, TXD2);

  Serial.println("===== SYSTEM START =====");

  // ===== HX711 =====
  scale.begin(
    LOADCELL_DOUT_PIN,
    LOADCELL_SCK_PIN
  );

  scale.set_scale();
  scale.tare();

  // ===== SHT31 =====
  Wire.begin(21, 22);

  if (!sht31.begin(SHT31_ADDR)) {

    Serial.println("❌ SHT31 non détecté !");

  } else {

    Serial.println("✅ SHT31 prêt !");
  }

  // ===== MPU9250 =====
  Wire1.begin(19, 23);

  IMU.setWire(&Wire1);

  IMU.beginAccel();

  Serial.println("✅ MPU9250 prêt !");

  // ===== WIFI =====
  connectWiFi();
}

// =====================================================
// ================= LOOP ==============================
// =====================================================
void loop() {

  // ===== WIFI =====
  if (WiFi.status() != WL_CONNECTED) {

    connectWiFi();
  }

  // ===== GET STATUS =====
  getExtractionStatus();

  // ===== HX711 =====
  scale.set_scale(calibration_factor);

  float weight = scale.get_units(10);

  filteredWeight =
      alpha * weight
      + (1 - alpha) * filteredWeight;

  weight = filteredWeight;

  if (abs(weight - lastWeight) > 0.05) {

    Serial.println("⚡ Variation détectée !");
  }

  lastWeight = weight;

  // ===== SHT31 =====
  float temp = sht31.readTemperature();

  float hum = sht31.readHumidity();

  if (isnan(temp) || isnan(hum)) {

    temp = 0;
    hum = 0;
  }

  // ===== MPU9250 =====
  IMU.accelUpdate();

  float ax = IMU.accelX();
  float ay = IMU.accelY();
  float az = IMU.accelZ();

  // ===== DISPLAY =====
  Serial.print("Poids: ");
  Serial.print(weight);

  Serial.print(" | Temp: ");
  Serial.print(temp);

  Serial.print(" | Hum: ");
  Serial.print(hum);

  Serial.print(" | Acc: ");

  Serial.print(ax);
  Serial.print(",");

  Serial.print(ay);
  Serial.print(",");

  Serial.println(az);

  // ===== BLUETOOTH =====
  hc06.print(weight);

  hc06.print(",");

  hc06.print(temp);

  hc06.print(",");

  hc06.println(hum);

  // ===== SEND ONLY IF EXTRACTION ACTIVE =====
  if (extractionON) {

    sendSensorData(
      weight,
      temp,
      hum,
      ax,
      ay,
      az
    );
  }

  delay(2000);
}
